Shader "GDD/Campania/PasoVientoHeladoHorizonte"
{
    Properties
    {
        _MainTex ("Horizonte", 2D) = "black" {}
        _Luz ("Luz ambiente", Range(0,1)) = 1
        _Scroll ("Desplazamiento horizontal", Float) = 0
        _Tinte ("Hora del dia", Color) = (1,1,1,1)
    }
    SubShader
    {
        // Se dibuja antes que toda la geometria y sin escribir profundidad: el mundo lo tapa naturalmente.
        Tags { "Queue"="Background+10" "RenderType"="Background" "IgnoreProjector"="True" }
        ZWrite Off
        // LEqual: el terreno y el lago (mas cerca) lo tapan sin importar el orden de dibujo.
        ZTest LEqual
        Cull Off
        Pass
        {
            CGPROGRAM
            #pragma vertex vert
            #pragma fragment frag
            #include "UnityCG.cginc"
            sampler2D _MainTex;
            float _Luz;
            float _Scroll;
            fixed4 _Tinte;
            struct appdata { float4 vertex : POSITION; float2 uv : TEXCOORD0; };
            struct v2f { float4 pos : SV_POSITION; float2 uv : TEXCOORD0; };
            v2f vert(appdata v)
            {
                v2f o;
                o.pos = UnityObjectToClipPos(v.vertex);
                o.uv = v.uv;
                return o;
            }
            fixed4 frag(v2f i) : SV_Target
            {
                // La cordillera ocupa el tercio superior; debajo continua la bruma del valle.
                float2 uv = float2(.05 + i.uv.x * .90 + _Scroll, saturate((i.uv.y - .72) / .28));
                fixed4 c = tex2D(_MainTex, uv);
                return fixed4(c.rgb * _Luz * _Tinte.rgb, 1);
            }
            ENDCG
        }
    }
}
